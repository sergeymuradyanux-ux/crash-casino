function drawCanvas(){ ctx.clearRect(0,0,gc.width,gc.height); const W=gc.width,H=gc.height; const pad={l:44,b:28,r:16,t:16}; const gW=W-pad.l-pad.r,gH=H-pad.t-pad.b; /* grid lines */ ctx.strokeStyle='rgba(255,255,255,0.04)';ctx.lineWidth=1; for(let i=0;i<=4;i++){ const y=pad.t+gH*(i/4); ctx.beginPath();ctx.moveTo(pad.l,y);ctx.lineTo(W-pad.r,y);ctx.stroke(); } for(let i=0;i<=4;i++){ const x=pad.l+gW*(i/4); ctx.beginPath();ctx.moveTo(x,pad.t);ctx.lineTo(x,H-pad.b);ctx.stroke(); } if(gameState==='idle'||gameState==='countdown') return; if(points.length<2) return; const maxM=Math.max(currentMult,1.5); const scX=gW/Math.max(points.length-1,1); const scY=gH/(maxM-1||1); const px=i=>pad.l+i*scX; const py=m=>pad.t+gH-(m-1)*scY; const crashed=gameState==='crashed'; const lc=crashed?'#ef4444':'#22c55e'; /* gradient fill under curve */ ctx.beginPath(); ctx.moveTo(px(0),py(1)); points.forEach((m,i)=>ctx.lineTo(px(i),py(m))); ctx.lineTo(px(points.length-1),H-pad.b); ctx.lineTo(px(0),H-pad.b); ctx.closePath(); const g=ctx.createLinearGradient(0,pad.t,0,H-pad.b); g.addColorStop(0,crashed?'rgba(239,68,68,0.35)':'rgba(34,197,94,0.3)'); g.addColorStop(0.6,crashed?'rgba(239,68,68,0.08)':'rgba(34,197,94,0.06)'); g.addColorStop(1,'rgba(0,0,0,0)'); ctx.fillStyle=g;ctx.fill(); /* main curve line — thick glowing */ ctx.beginPath(); ctx.moveTo(px(0),py(1)); points.forEach((m,i)=>ctx.lineTo(px(i),py(m))); ctx.strokeStyle=lc;ctx.lineWidth=3;ctx.lineJoin='round';ctx.lineCap='round'; ctx.shadowColor=lc;ctx.shadowBlur=12; ctx.stroke();ctx.shadowBlur=0; /* second thinner bright line on top for crispness */ ctx.beginPath(); ctx.moveTo(px(0),py(1)); points.forEach((m,i)=>ctx.lineTo(px(i),py(m))); ctx.strokeStyle=crashed?'#fca5a5':'#86efac';ctx.lineWidth=1.5; ctx.shadowBlur=0;ctx.stroke(); /* dot at current tip */ const tx=px(points.length-1),ty=py(points[points.length-1]); ctx.beginPath();ctx.arc(tx,ty,5,0,Math.PI*2); ctx.fillStyle=lc;ctx.shadowColor=lc;ctx.shadowBlur=14;ctx.fill();ctx.shadowBlur=0; ctx.beginPath();ctx.arc(tx,ty,2.5,0,Math.PI*2); ctx.fillStyle='#fff';ctx.fill(); /* y axis labels */ ctx.fillStyle='rgba(148,163,184,0.5)';ctx.font='600 10px sans-serif';ctx.textAlign='right'; for(let i=0;i<=4;i++){ const m=1+(maxM-1)*(i/4); ctx.fillText(m.toFixed(2)+'×',pad.l-6,py(m)+4); } }
